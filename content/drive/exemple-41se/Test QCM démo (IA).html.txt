// État de l'application
let activeQuestions = [];
let currentIndex = 0;
let userAnswers = {}; // { questionId: { selectedOriginalIdx, isCorrect } }
let studyMode = false;

function shuffleOptions(questionObj) {
    // Crée une copie mélangée des options tout en gardant la trace de la bonne réponse
    const mapped = questionObj.options.map((text, idx) => ({
        text,
        isCorrect: idx === questionObj.correct
    }));
    for (let i = mapped.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [mapped[i], mapped[j]] = [mapped[j], mapped[i]];
    }
    return mapped;
}

function initQuiz(questionsSubset = null) {
    const selectedCat = document.getElementById('categorySelect').value;
    let baseList = questionsSubset || MASTER_QUESTIONS;

    if (!questionsSubset && selectedCat !== 'ALL') {
        baseList = MASTER_QUESTIONS.filter(q => q.cat === selectedCat);
    }

    activeQuestions = baseList.map(q => ({
        ...q,
        shuffledOptions: shuffleOptions(q)
    }));

    currentIndex = 0;
    userAnswers = {};
    document.getElementById('resultsCard').classList.add('hidden');
    document.getElementById('studyCard').classList.add('hidden');
    document.getElementById('quizCard').classList.remove('hidden');
    document.getElementById('statsBar').classList.remove('hidden');
    document.getElementById('progressBarContainer').classList.remove('hidden');
    studyMode = false;
    document.getElementById('studyBtn').textContent = "📖 Mode Fiches de révision";

    renderQuestion();
    updateStats();
}

function filterByCategory() {
    initQuiz();
}

function renderQuestion() {
    const q = activeQuestions[currentIndex];
    document.getElementById('qCategory').textContent = q.catName;
    document.getElementById('qNumber').textContent = `Question ${currentIndex + 1} sur ${activeQuestions.length}`;
    document.getElementById('qText').textContent = q.q;

    const optionsList = document.getElementById('optionsList');
    optionsList.innerHTML = '';

    const letters = ['A', 'B', 'C', 'D'];
    const savedAnswer = userAnswers[q.id];

    q.shuffledOptions.forEach((opt, idx) => {
        const btn = document.createElement('button');
        btn.className = 'option-btn';
        btn.innerHTML = `<span class="option-letter">${letters[idx]}</span><span>${opt.text}</span>`;

        if (savedAnswer !== undefined) {
            btn.disabled = true;
            if (opt.isCorrect) {
                btn.classList.add('correct');
            } else if (savedAnswer.selectedIdx === idx) {
                btn.classList.add('wrong');
            }
        } else {
            btn.onclick = () => selectOption(idx);
        }

        optionsList.appendChild(btn);
    });

    const expBox = document.getElementById('explanationBox');
    if (savedAnswer !== undefined) {
        showExplanation(q, savedAnswer.isCorrect);
    } else {
        expBox.style.display = 'none';
    }

    document.getElementById('prevBtn').disabled = currentIndex === 0;
    document.getElementById('nextBtn').textContent =
        currentIndex === activeQuestions.length - 1 ? "Voir le bilan final 🏁" : "Suivant ➡";
}

function selectOption(idx) {
    const q = activeQuestions[currentIndex];
    const chosenOpt = q.shuffledOptions[idx];
    const isCorrect = chosenOpt.isCorrect;

    userAnswers[q.id] = {
        selectedIdx: idx,
        isCorrect: isCorrect
    };

    renderQuestion();
    updateStats();
}

function showExplanation(q, isCorrect) {
    const expBox = document.getElementById('explanationBox');
    const expTitle = document.getElementById('expTitle');
    expBox.style.display = 'block';
    expBox.style.borderLeftColor = isCorrect ? 'var(--success)' : 'var(--error)';

    expTitle.textContent = isCorrect ? "✅ Exact !" : "❌ Réponse incorrecte";
    expTitle.style.color = isCorrect ? 'var(--success)' : 'var(--error)';

    document.getElementById('expText').textContent = q.exp;
    document.getElementById('expQuote').textContent = q.quote;
}

function updateStats() {
    const answeredCount = Object.keys(userAnswers).length;
    const correctCount = Object.values(userAnswers).filter(a => a.isCorrect).length;
    const errorCount = answeredCount - correctCount;
    const percent = answeredCount > 0 ? Math.round((correctCount / answeredCount) * 100) : 0;

    document.getElementById('statProgress').textContent = `${currentIndex + 1} / ${activeQuestions.length}`;
    document.getElementById('statScore').textContent = correctCount;
    document.getElementById('statErrors').textContent = errorCount;
    document.getElementById('statPercent').textContent = `${percent}%`;

    const progressPercent = (answeredCount / activeQuestions.length) * 100;
    document.getElementById('progressFill').style.width = `${progressPercent}%`;
}

function prevQuestion() {
    if (currentIndex > 0) {
        currentIndex--;
        renderQuestion();
        updateStats();
    }
}

function nextQuestion() {
    if (currentIndex < activeQuestions.length - 1) {
        currentIndex++;
        renderQuestion();
        updateStats();
    } else {
        showResults();
    }
}

function showResults() {
    document.getElementById('quizCard').classList.add('hidden');
    document.getElementById('resultsCard').classList.remove('hidden');

    const total = activeQuestions.length;
    const correct = Object.values(userAnswers).filter(a => a.isCorrect).length;
    const wrongCount = total - correct;
    const pct = Math.round((correct / total) * 100);

    let msg = `Vous avez obtenu <strong>${correct} / ${total} (${pct}%)</strong>. `;
    if (pct === 100) {
        msg += `🏆 Félicitations ! Vous maîtrisez 100 % des informations du texte d'Isabelle Attané.`;
    } else {
        msg += `Il reste <strong>${wrongCount} question(s)</strong> non maîtrisée(s). Cliquez sur le bouton rouge ci-dessous pour retravailler uniquement vos erreurs jusqu'à atteindre 100 %.`;
    }
    document.getElementById('finalSummaryText').innerHTML = msg;

    // Détail par catégorie
    const breakdownDiv = document.getElementById('categoryBreakdown');
    breakdownDiv.innerHTML = '';

    const categories = {};
    activeQuestions.forEach(q => {
        if (!categories[q.cat]) {
            categories[q.cat] = { name: q.catName, total: 0, correct: 0 };
        }
        categories[q.cat].total++;
        if (userAnswers[q.id] && userAnswers[q.id].isCorrect) {
            categories[q.cat].correct++;
        }
    });

    Object.values(categories).forEach(c => {
        const cPct = Math.round((c.correct / c.total) * 100);
        const row = document.createElement('div');
        row.className = 'cat-row';
        row.innerHTML = `
            <span style="font-weight: 600;">${c.name}</span>
            <span style="font-weight: 800; color: ${cPct === 100 ? 'var(--success)' : 'var(--ined-dark)'}">
                ${c.correct} / ${c.total} (${cPct}%)
            </span>
        `;
        breakdownDiv.appendChild(row);
    });

    const retryBtn = document.getElementById('retryWrongBtn');
    if (wrongCount === 0) {
        retryBtn.classList.add('hidden');
    } else {
        retryBtn.classList.remove('hidden');
        retryBtn.textContent = `🎯 Rejouer uniquement les ${wrongCount} question(s) ratée(s) ou non répondues`;
    }
}

function retryWrongQuestions() {
    const wrongQuestions = activeQuestions.filter(q => !userAnswers[q.id] || !userAnswers[q.id].isCorrect);
    if (wrongQuestions.length > 0) {
        initQuiz(wrongQuestions);
    }
}

function restartQuiz() {
    initQuiz();
}

function toggleStudyMode() {
    studyMode = !studyMode;
    const studyCard = document.getElementById('studyCard');
    const quizCard = document.getElementById('quizCard');
    const resultsCard = document.getElementById('resultsCard');
    const studyBtn = document.getElementById('studyBtn');

    if (studyMode) {
        quizCard.classList.add('hidden');
        resultsCard.classList.add('hidden');
        studyCard.classList.remove('hidden');
        studyBtn.textContent = "📝 Retour au QCM interactif";
        renderStudyList();
    } else {
        studyCard.classList.add('hidden');
        quizCard.classList.remove('hidden');
        studyBtn.textContent = "📖 Mode Fiches de révision";
    }
}

function renderStudyList() {
    const selectedCat = document.getElementById('categorySelect').value;
    const list = selectedCat === 'ALL'
        ? MASTER_QUESTIONS
        : MASTER_QUESTIONS.filter(q => q.cat === selectedCat);

    const container = document.getElementById('studyList');
    container.innerHTML = '';

    list.forEach((q, idx) => {
        const div = document.createElement('div');
        div.className = 'study-item';
        div.innerHTML = `
            <div style="font-size: 0.8rem; color: var(--ined-blue); font-weight: bold;">${q.catName} — #${idx + 1}</div>
            <div style="font-weight: 700; font-size: 1.05rem; margin-top: 0.25rem;">${q.q}</div>
            <div class="study-ans">✔ Réponse : ${q.options[q.correct]}</div>
            <div style="font-size: 0.92rem;">${q.exp}</div>
            <div class="quote-box">${q.quote}</div>
        `;
        container.appendChild(div);
    });
}
