<script>
(function () {
  var ANSWERS = { q1:'b', q2:'a', q3:'c', q4:'a', q5:'a', q6:'b', q7:'c', q8:'a', q9:'a', q10:'a' };
  var FIX = {
    q1: 'To give up = abandonner, arrêter.',
    q2: 'Reliable = fiable.',
    q3: 'A deadline = une date limite.',
    q4: 'Eventually = finalement, à la fin.',
    q5: 'An achievement = une réussite.',
    q6: 'To look forward to = avoir hâte de.',
    q7: 'Although = even though = bien que.',
    q8: 'Accurate = précis, exact.',
    q9: 'A shortcut = un raccourci.',
    q10: 'To be about to = être sur le point de.'
  };
  var form = document.getElementById('qcm');

  function fieldOf(k) { return document.getElementById('fb' + k.slice(1)).closest('fieldset'); }

  function show(k, ok, answered) {
    var fb = document.getElementById('fb' + k.slice(1));
    var fs = fieldOf(k);
    fs.classList.remove('answered-ok', 'answered-ko');
    if (!answered) { fb.textContent = ''; fb.className = 'fb'; return; }
    fb.textContent = (ok ? '✔ Bonne réponse ! ' : '✘ Réponse incorrecte. ') + FIX[k];
    fb.className = 'fb ' + (ok ? 'ok' : 'ko');
    fs.classList.add(ok ? 'answered-ok' : 'answered-ko');
  }

  document.getElementById('check').addEventListener('click', function () {
    var score = 0, total = 0, k;
    for (k in ANSWERS) {
      total++;
      var sel = form.querySelector('input[name=' + k + ']:checked');
      var ok = !!sel && sel.value === ANSWERS[k];
      if (ok) score++;
      show(k, ok, true);
    }
    document.getElementById('score').textContent = 'Score : ' + score + ' / ' + total;
    window.scrollTo({ top: 0, behavior: 'smooth' });
  });

  document.getElementById('reset').addEventListener('click', function () {
    form.reset();
    for (var k in ANSWERS) show(k, false, false);
    document.getElementById('score').textContent = '';
    window.scrollTo({ top: 0, behavior: 'smooth' });
  });

  form.addEventListener('change', function (e) {
    var t = e.target;
    if (t && t.name && ANSWERS[t.name]) show(t.name, t.value === ANSWERS[t.name], true);
  });
})();
</script>
